$errors = [];

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    $password = $_POST['password'];
    // $login_type = ($_POST['login_type'] === 'admin') ? 'admin' : 'customer';
    $old_email = $email;

    if ($email === '' || $password === '') {
        $errors[] = "Please enter both email and password.";
    } else {
        $stmt = $conn->prepare("SELECT id, first_name, password, role FROM users WHERE email = ?");
        $stmt->bind_param("s", $email);
        $stmt->execute();
        $result = $stmt->get_result();

        if ($result->num_rows === 1) {
            $user = $result->fetch_assoc();

            if (password_verify($password, $user['password'])) {

                // Make sure the account role matches the tab they logged in from
                // if ($user['role'] !== $login_type) {
                //     if ($login_type === 'admin') {
                //         $errors[] = "This account is not an admin account. Please use the Customer tab.";
                //     } else {
                //         $errors[] = "This is an admin account. Please use the Admin tab.";
                //     }
                } else {
                    $_SESSION['user_id'] = $user['id'];
                    $_SESSION['first_name'] = $user['first_name'];
                    $_SESSION['role'] = $user['role'];

                    redirect($user['role'] === 'admin' ? '/kinmel/admin/index.php' : '/kinmel/home.php');
                }
            } else {
                $errors[] = "Incorrect email or password.";
            }
        // } else {
        //     $errors[] = "Incorrect email or password.";
        }
        $stmt->close();
    }

<script>
    const tabs = document.querySelectorAll('.login-tab');
    const loginTypeInput = document.getElementById('loginTypeInput');
    const submitBtn = document.getElementById('loginSubmitBtn');
    const customerLinks = document.getElementById('customerLinks');
    const socialDivider = document.getElementById('socialLoginDivider');
    const socialButtons = document.getElementById('socialLoginButtons');

    tabs.forEach(function (tab) {
        tab.addEventListener('click', function () {
            tabs.forEach(t => t.classList.remove('active'));
            tab.classList.add('active');

            const type = tab.getAttribute('data-tab');
            loginTypeInput.value = type;

            if (type === 'admin') {
                submitBtn.textContent = 'Login as Admin';
                customerLinks.style.display = 'none';
                socialDivider.style.display = 'none';
                socialButtons.style.display = 'none';
            } else {
                submitBtn.textContent = 'Login as Customer';
                customerLinks.style.display = 'block';
                socialDivider.style.display = 'block';
                socialButtons.style.display = 'flex';
            }
        });
    });
